#!/usr/bin/env bash
# Strict start-on-anchor citation check.
# usage: citecheck.sh TARGET_FILE RECORDS_FILE
# record: label|start|end|start_anchor|end_anchor|claim
#   anchors/claim are fixed substrings; a leading "=" means exact whole-line match;
#   "-" means "not checked".
set -u
T="$1"; R="$2"
[ -s "$T" ] || { echo "VOID: target $T missing/empty"; exit 2; }
[ -s "$R" ] || { echo "VOID: records $R missing/empty"; exit 2; }
total=$(wc -l < "$T")
fails=0; n=0
match() { # line pattern
  local line="$1" pat="$2"
  if [ "${pat:0:1}" = "=" ]; then [ "$line" = "${pat:1}" ]; else [[ "$line" == *"$pat"* ]]; fi
}
while IFS='|' read -r label s e sa ea claim; do
  [ -z "$label" ] && continue
  case "$label" in \#*) continue;; esac
  n=$((n+1)); why=""
  if [ "$s" -gt "$total" ] || [ "$e" -gt "$total" ]; then why="range $s-$e beyond EOF ($total)"; else
    sl=$(sed -n "${s}p" "$T"); el=$(sed -n "${e}p" "$T"); rng=$(sed -n "${s},${e}p" "$T")
    match "$sl" "$sa" || why="$why start line $s does not match anchor [$sa] (is: ${sl:0:70});"
    if [ "$ea" != "-" ]; then match "$el" "$ea" || why="$why end line $e does not match [$ea] (is: ${el:0:70});"; fi
    if [ "$claim" != "-" ]; then [[ "$rng" == *"$claim"* ]] || why="$why range $s-$e lacks claim [$claim];"; fi
  fi
  if [ -z "$why" ]; then echo "PASS  $label :$s-$e"; else echo "FAIL  $label :$s-$e --$why"; fails=$((fails+1)); fi
done < "$R"
[ "$n" -gt 0 ] || { echo "VOID: no records read"; exit 2; }
echo "records=$n fails=$fails"
[ "$fails" -eq 0 ]
