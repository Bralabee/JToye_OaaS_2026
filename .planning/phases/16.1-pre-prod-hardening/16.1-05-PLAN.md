    import org.junit.jupiter.api.Tag;
    import org.junit.jupiter.api.Test;
    import org.springframework.beans.factory.annotation.Autowired;
    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.jdbc.core.JdbcTemplate;
    import org.springframework.test.context.ActiveProfiles;
    import org.springframework.test.context.DynamicPropertyRegistry;
    import org.springframework.test.context.DynamicPropertySource;
    import org.testcontainers.containers.PostgreSQLContainer;
    import org.testcontainers.junit.jupiter.Container;
    import org.testcontainers.junit.jupiter.Testcontainers;

    import java.util.List;
    import java.util.Map;
    import java.util.Set;

    import static org.assertj.core.api.Assertions.assertThat;

    /**
     * AUDIT-W0-05 contract test: every regular table in the public schema must have
     * both ROW LEVEL SECURITY enabled AND forced. Catches the recurring failure mode
     * where a new migration adds a tenant-scoped table without RLS (root cause of
     * V11→V4, V14→V9, V15→V5, V33→V27/28/29 retro-patches per
     * docs/audit/remediation/03-database-remediation.md Finding 11).
     */
    @SpringBootTest
    @Testcontainers
    @ActiveProfiles("test")
    @Tag("testcontainers")
    class RlsContractTest {

        @Container
        static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:15")
                .withDatabaseName("jtoye_test")
                .withUsername("test")
                .withPassword("test");

        @DynamicPropertySource
        static void props(DynamicPropertyRegistry r) {
            r.add("spring.datasource.url", postgres::getJdbcUrl);
            r.add("spring.datasource.username", postgres::getUsername);
            r.add("spring.datasource.password", postgres::getPassword);
            r.add("spring.flyway.enabled", () -> "true");
            r.add("rate-limiting.enabled", () -> "false");
        }

        @Autowired private JdbcTemplate jdbc;

        /**
         * Tables in the public schema that are intentionally NOT tenant-scoped and
         * therefore not RLS-controlled. Every entry needs a written justification.
         */
        private static final Set<String> EXEMPT_TABLES = Set.of(
            // Flyway migration metadata; managed by Flyway itself.
            "flyway_schema_history",
            // AUDIT-W0-03 idempotency log; Stripe event_ids are globally unique,
            // table is infrastructure-level, NOT tenant data. Per V35 comment +
            // 16.1-CONTEXT.md <decisions> Item 3 LOCKED: "NOT RLS-enabled".
            "processed_stripe_events"
        );

        @Test
        void everyPublicTableHasRlsAndForce() {
            List<Map<String, Object>> tables = jdbc.queryForList(
                "SELECT relname, relrowsecurity, relforcerowsecurity " +
                "FROM pg_class " +
                "WHERE relkind = 'r' " +
                "  AND relnamespace = 'public'::regnamespace " +
                "ORDER BY relname");

            for (Map<String, Object> row : tables) {
                String name = (String) row.get("relname");
                if (EXEMPT_TABLES.contains(name)) continue;

                assertThat(row.get("relrowsecurity"))
                    .as("ENABLE ROW LEVEL SECURITY missing on public.%s — every tenant-scoped table " +
                        "must enable RLS in its Flyway migration. If %s is intentionally not " +
                        "tenant-scoped, add it to RlsContractTest.EXEMPT_TABLES with a written " +
                        "justification.", name, name)
                    .isEqualTo(true);

                assertThat(row.get("relforcerowsecurity"))
                    .as("FORCE ROW LEVEL SECURITY missing on public.%s — table-owner / superuser " +
                        "would bypass tenant isolation. Add `ALTER TABLE %s FORCE ROW LEVEL SECURITY` " +
                        "to the migration that creates the table.", name, name)
                    .isEqualTo(true);
            }
        }

        @Test
        void auditW0_05_targetTablesAreForced() {
            List<String> w0_05_targets = List.of(
                "reviews", "shop_promotions", "shop_announcements",
                "customers_aud", "shops_aud", "products_aud",
                "financial_transactions_aud", "orders_aud", "order_items_aud"
            );

            for (String t : w0_05_targets) {
                Boolean forced = jdbc.queryForObject(
                    "SELECT relforcerowsecurity FROM pg_class " +
                    "WHERE relkind = 'r' AND relnamespace = 'public'::regnamespace " +
                    "  AND relname = ?",
                    Boolean.class, t);
                assertThat(forced)
                    .as("AUDIT-W0-05 target %s missing FORCE ROW LEVEL SECURITY — V35 must add it", t)
                    .isNotNull()
                    .isEqualTo(true);
            }
        }

        @Test
        void noPolicyReadsBuggyAppTenantIdGuc() {
            // AUDIT-W0-04 sentinel: ensure no policy in the database still references
            // the buggy GUC name `app.tenant_id` (replaced by V35 with `app.current_tenant_id`).
            // This guards against a future migration accidentally re-introducing the typo.
            List<String> bad = jdbc.queryForList(
                "SELECT polname || ' on ' || polrelid::regclass::text " +
                "FROM pg_policy " +
                "WHERE pg_get_expr(polqual, polrelid)      LIKE '%app.tenant_id%' " +
                "   OR pg_get_expr(polwithcheck, polrelid) LIKE '%app.tenant_id%'",
                String.class);
            assertThat(bad)
                .as("policies referencing forbidden GUC `app.tenant_id` (canonical is `app.current_tenant_id`)")
                .isEmpty();
        }
    }

    The test exercises raw SQL INSERTs directly through `JdbcTemplate` so we control the exact GUC state per case (no need to spin up the storefront request context). Each test sets specific `app.current_tenant_id` and/or `app.customer_email` via `set_config(name, value, true)` (the canonical local-only setter used elsewhere in the codebase) and asserts whether the INSERT succeeds or fails.

    Five behavioural cases:

    1. `legitimateInsertWithMatchingOrderSucceeds` — set `app.current_tenant_id` AND `app.customer_email`; pre-seed a matching `orders` row owned by that tenant + email; INSERT a review with the same `tenant_id` + `order_id` + `customer_email`. Should succeed.

    2. `appBranchInsertWithMatchingTenantSucceeds` — set ONLY `app.current_tenant_id` (no customer_email); INSERT review with `tenant_id` matching. Should succeed (the app branch of the WITH CHECK fires).

    3. `spamReviewWithArbitraryTenantIsBlocked` — set `app.current_tenant_id` to UUID_A but `app.customer_email` to `attacker@evil.com`; attempt to INSERT a review with `tenant_id = UUID_B` (≠ A) and `customer_email = attacker@evil.com` and an `order_id` that does NOT exist OR exists under a different tenant/email. Should fail (RLS policy violation — both branches reject: app branch fails on tenant mismatch, customer branch fails on the EXISTS clause).

    4. `customerEmailMismatchIsBlocked` — set `app.customer_email = alice@example.com`; pre-seed an order owned by `bob@example.com`; INSERT review claiming `customer_email = bob@example.com` and the order's id. Should fail (the EXISTS subquery requires `o.customer_email = current_setting('app.customer_email')`).

    5. `nullCustomerEmailGucShortCircuitsCustomerBranch` — set ONLY `app.current_tenant_id` (no customer_email at all); INSERT review with `tenant_id` MISMATCHING the GUC. Should fail (app branch fails on tenant mismatch; customer branch short-circuits to false because the IS NOT NULL guard rejects an unset GUC).

     *   - allow the legitimate write path (matching app.current_tenant_id OR matching
     *     app.customer_email + EXISTS proof against the orders table);
     *   - block the spam-review write hole that V27's OR-clause permitted.

        private UUID otherTenantId;
        private UUID shopId;
        private UUID otherShopId;
        private UUID orderId;
        private UUID otherOrderId;
